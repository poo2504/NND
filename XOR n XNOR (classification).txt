import numpy as np
from tensorflow.keras.models import Sequential
from tensorflow.keras.layers import Dense, Input
from tensorflow.keras.optimizers import Adam

def classification_model(x, y, learning_rate, epochs):
    model = Sequential([
        Input(shape=(x.shape[1],)),
        Dense(4, activation='relu'),
        Dense(4, activation='relu'),
        Dense(1, activation='sigmoid')
    ])
    model.compile(optimizer=Adam(learning_rate=learning_rate),
                  loss="binary_crossentropy",
                  metrics=['accuracy'])
    model.fit(x, y, epochs=epochs, verbose=0)
    return model

# XNOR Gate (Same = 1, Different = 0)
def xnor_gate():
    x = np.array([[0,0],[0,1],[1,0],[1,1]])
    y = np.array([[1],[0],[0],[1]])
    return x, y

# XOR Gate (Different = 1, Same = 0)
def xor_gate():
    x = np.array([[0,0],[0,1],[1,0],[1,1]])
    y = np.array([[0],[1],[1],[0]])
    return x, y

def predictions(x, y, model):
    pred_values = model.predict(x)
    for actual, pred in zip(y, pred_values):
        binary = 1 if pred > 0.5 else 0
        print(f'Actual Value: {actual[0]}, Predicted (raw): {pred[0]:.4f}, Binary: {binary}')

# Training and prediction for XNOR gate
print('Training for XNOR gate: ')
x, y = xnor_gate()
model = classification_model(x, y, 0.1, 1000)
predictions(x, y, model)

# Training and prediction for XOR gate
print('\nTraining for XOR gate: ')
x, y = xor_gate()
model = classification_model(x, y, 0.1, 1000)
predictions(x, y, model)
